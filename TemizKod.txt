.model small
.stack 100h

.data
    ; --- OYUN DEGISKENLERI ---
    t_puan      dw 0          ; Tur boyunca kazanilan toplam puanin tutuldugu degisken
    c_hiz       db 4          ; Carkin donus hizini tutar (Kucuk deger = Hizli)
    cark_idx    db 0          ; Carkin o anki secili diliminin indeksini (0-7 arasi) tutar
    last_time   db 0          ; Zamanlayici (timer) icin son okunan zaman degerini tutar
    wait_cnt    db 0          ; Hiz kontrolu icin bekleme sayaci
    slot_tmp    db 0          ; Ekrana cark cizilirken kullanilan gecici dongu degiskeni
    spin_pos    db 0          ; Ortadaki donen animasyon karakterinin donus pozisyonu

    ; --- HAK VE REKOR DEGISKENLERI ---
    hak         db 3          ; Oyuncunun baslangic hakki (3)
    h_puan      dw 0          ; Dosyadan okunan ve dosyaya yazilan en yuksek rekor puan
    dosya_adi   db 'REKOR.DAT', 0 ; Rekorun kaydedilecegi dosyanin adi (sonu 0 ile bitmeli)
    handle      dw 0          ; Dosya acildiginda isletim sisteminin verdigi dosya numarasi
    
    ; --- EKRAN MESAJLARI ---
    rmsg        db 'En Yuksek Skor: $'
    pmsg        db 'Toplam Puan   : $'
    hmsg        db 'Kalan Hak     : $'
    m1          db '--- ANA MENU ---$'
    m2          db '1) Oyuna Basla (Hiz Rastgele)$'
    m5          db 'Secim (1) - Cikis(Q/ESC): $'

    ; INT 10h ile renklendirilecegi icin sonunda '$' yok, tam 24 karakter.
    baslik      db '--- DONEN CARK OYUNU ---' 
    bilgi       db 'Durdurmak icin SPACE tusuna basin.$'
    
    msg_kazan1  db '>>> $'
    msg_kazan2  db ' PUAN KAZANDINIZ! <<<      $'
    
    iflas       db '>>> IFLAS! PUAN SIFIRLANDI. <<<    $'
    tekrar      db 'Tekrar(E) / Menu(ESC) / Cikis(Q)   $'
    bitti_m     db '>>> HAK BITTI! MENU(ESC) / CIKIS(Q)$'
    temiz_m     db '                                     $' ; Ekranda onceki yazilari silmek icin bosluk
    temiz_p     db '          $'                               ; Rakamlari silmek icin kucuk bosluk

    ; Ortadaki ok animasyonunun karakterleri
    spin_tab    db '|', '/', '-', '\'
    
    ; --- 8 ADET CARK DILIMI YAZISI ---
    s0 db '[ 100 ]'
    s1 db '[ 250 ]'
    s2 db '[IFLAS]'
    s3 db '[  50 ]'
    s4 db '[ 500 ]'
    s5 db '[IFLAS]'
    s6 db '[ 200 ]'
    s7 db '[1000 ]' 

    ; Dilimlerin adreslerini, puan degerlerini ve renk kodlarini tutan diziler
    str_ptr dw offset s0, offset s1, offset s2, offset s3, offset s4, offset s5, offset s6, offset s7
    p_val   dw 100, 250, 0, 50, 500, 0, 200, 1000
    col_val db 0Bh, 0Ah, 0Ch, 0Eh, 0Dh, 0Ch, 09h, 03h 
    
    ; Yazilarin ekranda duracagi Video RAM (VRAM) bellek koordinatlari
    d_vram  dw 1032, 1216, 1864, 2496, 2632, 2448, 1800, 1168
    
    ; --- CARK DAIRESININ KOORDINATLARI (Eski, Duzgun Calisan 18 Noktali Versiyon) ---
    c_row db 6,6,7,7,8,8,9,9,11,11,13,13,15,15,16,16,17,17  ; Noktalarin satir numaralari
    c_col db 33,47,30,50,28,52,27,53,26,54,26,54,27,53,30,50,33,47  ; Noktalarin sutun numaralari
    c_cnt db 18                                             ; Toplam cizilecek nokta sayisi

.code
main proc
    mov ax, @data       ; Data segmentin adresini AX'e al
    mov ds, ax          ; AX'teki adresi DS (Data Segment) registerina yukle
    
    mov ax, 0003h       ; BIOS Video kesmesi icin AX'e 0003h (80x25 renkli text modu) yukle
    int 10h             ; Video modunu ayarla (Ekran temizlenir)
    
    call dosya_oku      ; Oyun baslamadan once REKOR.DAT dosyasindan rekoru oku

ana_menu:
    call ekran_temizle  ; Menuyu cizmeden once ekrani temizle
    mov t_puan, 0       ; Yeni oyunda toplam puani sifirla
    
    ; Ana menu metinlerini ekrana yazdirma islemleri
    mov dh, 6           ; Satir (Y) koordinati
    mov dl, 30          ; Sutun (X) koordinati
    call gotoxy         ; Imleci bu koordinata tasi
    lea dx, m1          ; DX'e menunun baslik metnini yukle
    call print_dos      ; Metni ekrana yazdir
    
    mov dh, 9
    mov dl, 24
    call gotoxy
    lea dx, m2
    call print_dos
    
    mov dh, 14
    mov dl, 26
    call gotoxy
    lea dx, m5
    call print_dos
    
    mov dh, 18
    mov dl, 28
    call gotoxy
    lea dx, rmsg
    call print_dos
    call rekor_guncelle ; Ekrana rekor puanin sayisal degerini yazdir

menu_tus:
    mov ah, 00h         ; Klavyeden tus okuma kesmesi fonksiyonu
    int 16h             ; Tusa basilmasini bekle ve AL registerina ASCII kodunu al
    
    cmp al, '1'         ; Eger '1' tusuna basildiysa...
    je oyunu_baslat     ; ... oyunu_baslat etiketine atla
    cmp al, 'q'         ; Eger 'q' tusuna basildiysa...
    je cikis            ; ... cikis'a atla
    cmp al, 'Q'         ; Eger 'Q' tusuna basildiysa...
    je cikis            ; ... cikis'a atla
    cmp al, 27          ; Eger ESC (ASCII 27) tusuna basildiysa...
    je cikis            ; ... cikis'a atla
    jmp menu_tus        ; Baska tusa basildiysa donguye devam et

oyunu_baslat:
    mov hak, 3          ; Oyun baslarken hakki 3 olarak ayarla
    jmp oyun_basla      ; Oyun dongusune gec

oyun_basla:
    call ekran_temizle  ; Carki cizmeden once ekrani temizle
    call sabit_ekran_ciz; Carkin dairesini, basliklari ve metinleri ekrana yerlestir
    
    ; --- RASTGELE HIZ BELIRLEME ---
    mov ah, 2Ch         ; Sistem saatini okuma fonksiyonu
    int 21h             ; DL registerina salise degerini getir
    mov al, dl          ; Saliseyi AL'ye kopyala
    and al, 03h         ; AL'nin sadece son 2 bitini al (0, 1, 2 veya 3 sonucu cikar)
    
    cmp al, 0           ; Eger sonuc 0 ise...
    je rng_hizli        ; ... hizli moda gec
    cmp al, 1           ; Eger sonuc 1 ise...
    je rng_normal       ; ... normal moda gec
    
    mov c_hiz, 25       ; Degilse cok yavas mod olsun (25 bekleme suresi)
    jmp rng_tamam
rng_hizli:
    mov c_hiz, 0        ; Hizli mod (Bekleme yok)
    jmp rng_tamam
rng_normal:
    mov c_hiz, 6        ; Normal mod (6 birim bekleme)
rng_tamam:

    call carki_ciz      ; Carki hesaplanan baslangic durumuyla ciz
    call klavye_temizle ; Buffer'da kalan eski tus basimlarini sil
    mov spin_pos, 0     ; Ortadaki okun pozisyonunu sifirla
    
    mov ah, 2Ch         ; Hizi kontrol etmek icin saati tekrar oku
    int 21h
    mov last_time, dl   ; Su anki saliseyi kaydet
    mov wait_cnt, 0     ; Bekleme sayacini sifirla

oyun_loop:
    call hizli_space_kontrol ; Bosluk (Space) tusuna basildi mi kontrol et
    cmp al, 1           ; Eger AL 1 ise basildi demektir
    je cark_durdu       ; Tusa basildiysa carki durdurma islemine gec
    
    cmp c_hiz, 0        ; Hiz 0 (en hizli) ise bekleme yapma
    je animasyon_adimi  ; Direkt animasyonu dondurmeye gec
    
    ; Zaman gecis kontrolu (Bekleme mekanizmasi)
    mov ah, 2Ch         
    int 21h
    cmp dl, last_time   ; Son okunan zaman ile su anki zaman ayni mi?
    je oyun_loop        ; Ayniysa donguye devam (zaman gecmemis)
    
    mov last_time, dl   ; Zaman degistiyse yeni zamani kaydet
    mov al, wait_cnt    ; Sayaci 1 artir
    inc al
    mov wait_cnt, al
    cmp al, c_hiz       ; Sayac hedeflenen hiza ulasti mi?
    jb oyun_loop        ; Ulasmadiysa beklemeye devam et
    mov wait_cnt, 0     ; Ulastiysa sayaci sifirla ve asagiya (animasyona) gec

animasyon_adimi:
    mov al, spin_pos    ; Ortadaki okun pozisyonunu al
    inc al              ; Bir sonraki sekle gec
    and al, 03h         ; 4 farkli sekil var (0,1,2,3), bit maskeleme ile sinirla
    mov spin_pos, al    ; Yeni sekli kaydet
    
    mov al, cark_idx    ; Carkin donus yonunu belirleyen indeks
    add al, 7           ; Carki saat yonunde dondurmek icin indeksi ilerlet
    cmp al, 8           ; 8 dilim var, 8'e ulastiysa...
    jb idx_ok
    sub al, 8           ; ... 8 cikarip basa (0'a) sar (Mod alma islemi)
idx_ok:
    mov cark_idx, al    ; Yeni indeksi kaydet
    call carki_ciz      ; Carki ekranda bir adim donmus sekilde yeniden ciz
    jmp oyun_loop       ; Tus basilana kadar sonsuz donmeye devam et

cark_durdu:
    dec hak             ; Cark durdugu icin oyuncunun hakkini 1 azalt
    xor ah, ah          ; AH'yi sifirla
    mov al, cark_idx    ; Durdugu dilimin indeksini AL'ye al
    shl ax, 1           ; Diziler word (2 byte) boyutunda oldugu icin indeksi 2 ile carp
    mov si, ax          ; SI dizin kaydedicisine indeksi at
    mov ax, p_val[si]   ; O dilimin puan degerini p_val dizisinden AX'e cek
    
    push ax             ; Puan degeri asagida bozulmasin diye stack'e (yigina) at
    
    cmp ax, 0           ; Eger puan 0 ise...
    je iflas_durumu     ; ... bu IFLAS dilimidir, iflasa git
    
    add t_puan, ax      ; Puan 0 degilse, toplam puana ekle

    ; --- REKOR KONTROLU (Aninda Guncelleme) ---
    mov bx, t_puan      ; Toplam puani BX'e al
    cmp bx, h_puan      ; Toplam puani rekor puanla karsilastir
    jbe rekor_gecilmedi ; Eger kucuk veya esitse (rekor gecilmediyse) atla
    mov h_puan, bx      ; Gecildiyse yeni rekoru hafizaya al
    call dosya_yaz      ; Yeni rekoru REKOR.DAT dosyasina kaydet
    call rekor_guncelle ; Ekranda rekor yazisini aninda guncelle
rekor_gecilmedi:

    ; Kazanma mesajini ekrana yazdir
    mov dh, 21
    mov dl, 23
    call gotoxy
    lea dx, temiz_m     ; Once o satiri temizle
    call print_dos
    mov dh, 21
    mov dl, 23
    call gotoxy
    
    lea dx, msg_kazan1  ; ">>> " yazisini cikar
    call print_dos
    
    pop ax              ; Stack'teki kazanilan puani geri cek
    call sayi_yazdir    ; Puani ekrana rakam olarak yazdir
    
    lea dx, msg_kazan2  ; " PUAN KAZANDINIZ! <<<" yazisini cikar
    call print_dos
    
    jmp tur_bitti       ; Tur bitti islemine gec
    
iflas_durumu:
    pop ax              ; Stack'i temizlemek icin bossa cikar (kullanilmayacak)
    mov t_puan, 0       ; IFLAS edildi, toplam puani sifirla
    mov dh, 21
    mov dl, 23
    call gotoxy
    lea dx, temiz_m     ; Ekrandaki satiri temizle
    call print_dos
    mov dh, 21
    mov dl, 23
    call gotoxy
    lea dx, iflas       ; "IFLAS! PUAN SIFIRLANDI" metnini yazdir
    call print_dos
    
tur_bitti:
    call puan_guncelle  ; Guncel puani ekrana yansit
    call hak_guncelle   ; Kalan hakki ekrana yansit
    
    mov dh, 22
    mov dl, 23
    call gotoxy
    lea dx, temiz_m
    call print_dos
    mov dh, 22
    mov dl, 23
    call gotoxy
    
    cmp hak, 0          ; Haklar tamamen bitti mi?
    je hak_bitti_mesaji ; Bittiyse oyun bitti mesajina atla
    lea dx, tekrar      ; Bitmediyse "Tekrar(E) / Menu(ESC)" yazdir
    call print_dos
    jmp son_tus

hak_bitti_mesaji:
    mov dh, 22
    mov dl, 23
    call gotoxy
    lea dx, bitti_m     ; "HAK BITTI" mesajini yazdir
    call print_dos

son_tus:
    mov ah, 00h         ; Klavyeden tus bekleme fonksiyonu
    int 16h
    
    cmp hak, 0          ; Hak 0 ise tekrar baslamaya izin verme
    je hak_yok_tus_kontrol
    cmp al, 'e'         ; 'e' basildiysa
    je oyun_basla       ; Yeni tur baslat
    cmp al, 'E'
    je oyun_basla
hak_yok_tus_kontrol:
    cmp al, 27          ; ESC basildiysa
    je ana_menu         ; Menuye don
    cmp al, 'q'         ; 'q' basildiysa
    je cikis            ; Oyundan cik
    cmp al, 'Q'
    je cikis
    jmp son_tus         ; Gecersiz tussa tekrar bekle

cikis:
    mov ah, 4Ch         ; DOS programdan cikis fonksiyonu
    int 21h             ; Isletim sistemine geri don
main endp

; --- EKRAN RENKLENDIRME PROSEDURU ---
renkli_yazi proc
    ; BP = Metnin hafizadaki adresi
    ; CX = Kac karakter yazilacagi
    ; BL = Renk kodu (orn. 0Ch Kirmizi)
    ; DH/DL = Yazdirilacak Satir ve Sutun
    push ax
    push bx
    push cx
    push dx
    push bp
    mov ax, @data
    mov es, ax          ; ES registerine data segmentini yukle (INT 10h bunu gerektirir)
    mov ah, 13h         ; Metin dizisini ekrana yazdirmak icin BIOS Video fonksiyonu
    mov al, 00h         ; Sadece karakterleri yazdir, imleci oynatma
    mov bh, 0           ; Video sayfasi 0
    int 10h             ; Kesmeyi cagir, ekranda renklendirerek yazsin
    pop bp
    pop dx
    pop cx
    pop bx
    pop ax
    ret
renkli_yazi endp

; --- DOSYA OKUMA PROSEDURU ---
dosya_oku proc
    push ax 
    push bx 
    push cx 
    push dx
    mov ah, 3Dh         ; DOS dosya acma fonksiyonu
    mov al, 00h         ; Sadece okuma modunda (Read Only) ac
    lea dx, dosya_adi   ; Acilacak dosyanin adi
    int 21h
    jc oku_son          ; Eger hata verirse (dosya yoksa) atla
    mov handle, ax      ; Basariliysa dosya kimligini (handle) kaydet
    
    mov ah, 3Fh         ; DOS dosyadan veri okuma fonksiyonu
    mov bx, handle      ; Kimligi BX'e ver
    mov cx, 2           ; 2 byte (1 Word, rekor puani boyutu) oku
    lea dx, h_puan      ; Okunan veriyi h_puan degiskenine yaz
    int 21h
    
    mov ah, 3Eh         ; DOS dosya kapatma fonksiyonu
    mov bx, handle
    int 21h             ; Dosyayi guvenlice kapat
oku_son:
    pop dx 
    pop cx 
    pop bx 
    pop ax
    ret
dosya_oku endp

; --- DOSYA YAZMA PROSEDURU ---
dosya_yaz proc
    push ax 
    push bx 
    push cx 
    push dx
    mov ah, 3Ch         ; DOS dosya olusturma/ustune yazma fonksiyonu
    mov cx, 0           ; Dosya ozelligi (0 = normal dosya)
    lea dx, dosya_adi
    int 21h
    jc yaz_son          ; Hata olursa islemden cik
    mov handle, ax
    
    mov ah, 40h         ; DOS dosyaya veri yazma fonksiyonu
    mov bx, handle
    mov cx, 2           ; 2 byte'lik rekor bilgisini yazacagiz
    lea dx, h_puan      ; Hangi degiskenden yazacagimizi goster
    int 21h
    
    mov ah, 3Eh         ; Dosyayi kapat
    mov bx, handle
    int 21h
yaz_son:
    pop dx 
    pop cx 
    pop bx 
    pop ax
    ret
dosya_yaz endp

; --- REKOR EKRANINI GUNCELLEME ---
rekor_guncelle proc
    push ax 
    push bx 
    push cx 
    push dx
    mov dh, 18
    mov dl, 44          
    call gotoxy         ; Puanin oldugu koordinata git
    lea dx, temiz_p     ; Oradaki eski yaziyi bosluklarla sil
    call print_dos
    mov dh, 18
    mov dl, 44
    call gotoxy
    mov ax, h_puan      ; Ekrana yazilacak yeni rekoru AX'e al
    call sayi_yazdir    ; Rakamlari tek tek hesaplayip bas
    pop dx 
    pop cx 
    pop bx 
    pop ax
    ret
rekor_guncelle endp

; --- PUAN EKRANINI GUNCELLEME ---
puan_guncelle proc
    push ax 
    push bx 
    push cx 
    push dx
    mov dh, 19
    mov dl, 44          
    call gotoxy
    lea dx, temiz_p
    call print_dos
    mov dh, 19
    mov dl, 44
    call gotoxy
    mov ax, t_puan      ; T_puan degerini ekrana bas
    call sayi_yazdir
    pop dx 
    pop cx 
    pop bx 
    pop ax
    ret
puan_guncelle endp

; --- KALAN HAK EKRANINI GUNCELLEME ---
hak_guncelle proc
    push ax 
    push dx
    mov dh, 20
    mov dl, 44          
    call gotoxy
    mov al, hak         ; Hakki AL'ye al (Orn: 3)
    add al, '0'         ; Rakam olan 3'u, ASCII karakteri olan '3'e cevir (0'in ASCII degeri eklenir)
    mov dl, al          ; Yazilacak karakteri DL'ye koy
    mov ah, 02h         ; DOS tek karakter yazdirma fonksiyonu
    int 21h
    pop dx 
    pop ax
    ret
hak_guncelle endp

; --- SAYIYI STRING'E CEVIRIP YAZDIRMA ---
sayi_yazdir proc
    ; AX registerindeki sayisal degeri ekrana yazdirir
    cmp ax, 0           ; Sayi sifir mi?
    jne n_div           ; Sifir degilse bolme islemine basla
    mov dl, '0'         ; Sifirsa direkt ekrana '0' basip cik
    mov ah, 02h
    int 21h
    ret
n_div:
    xor cx, cx          ; CX sayacini sifirla (Kac basamak oldugunu tutacak)
    mov bx, 10          ; Sayiyi onluk tabana (10'a) bolecegiz
n1:
    xor dx, dx          ; Bolme isleminde DX sifirlanmali (kalan buraya gelecek)
    div bx              ; AX'i 10'a bol. (AX = Bolum, DX = Kalan/Son basamak)
    push dx             ; Kalan basamagi yigina (Stack) at (sayilar tersten birikir)
    inc cx              ; Basamak sayacini 1 artir
    cmp ax, 0           ; Bolum sifir oldu mu? (Sayi bitti mi?)
    jne n1              ; Bitmediyse donguye devam et
n2:
    pop dx              ; En son yigina atilan basamagi (en buyuk basamagi) geri cek
    add dl, '0'         ; Rakam degerini ekranda gosterebilmek icin ASCII karaktere cevir
    mov ah, 02h         ; DOS karakter yazdirma fonksiyonu
    int 21h             ; Ekrana bas
    loop n2             ; CX kadar (basamak sayisi kadar) islemi tekrar et
    ret
sayi_yazdir endp

; --- EKRAN TEMIZLEME ---
ekran_temizle proc
    mov ax, 0600h       ; Ekran kaydirma (temizleme) fonksiyonu
    mov bh, 07h         ; Siyah arka plan uzeri beyaz yazi renk kodu
    mov cx, 0000h       ; Sol ust kose koordinatlari (Satir 0, Sutun 0)
    mov dx, 184Fh       ; Sag alt kose koordinatlari (Satir 24, Sutun 79)
    int 10h             ; Tum sayfayi temizle
    ret
ekran_temizle endp

; --- IMLECI TASIMA ---
gotoxy proc
    ; DH = Satir, DL = Sutun
    push ax 
    push bx
    mov ah, 02h         ; BIOS imlec pozisyonu ayarlama
    mov bh, 0           ; Sayfa 0
    int 10h
    pop bx 
    pop ax
    ret
gotoxy endp

; --- DOS STRING YAZDIRMA ---
print_dos proc
    ; DX = Dizi adresi (Sonunda '$' olmali)
    mov ah, 09h         ; String yazdirma fonksiyonu
    int 21h
    ret
print_dos endp

; --- KLAVYE BUFFER TEMIZLEME ---
klavye_temizle proc
k1:
    mov ah, 01h         ; Klavyede basili tus var mi kontrol et
    int 16h
    jz k2               ; Yoksa (Zero Flag set ise) islemi bitir (k2'ye atla)
    mov ah, 00h         ; Varsa tusu oku ve hafizadan sil
    int 16h
    jmp k1              ; Diger tuslari silmek icin basa don
k2: ret
klavye_temizle endp

; --- SPACE TUSU HIZLI KONTROL ---
hizli_space_kontrol proc
    mov ah, 01h         ; Tus var mi diye bak (oyunu duraklatmadan)
    int 16h          
    jz sp_no            ; Yoksa sp_no'ya atla
    mov ah, 00h         ; Varsa tusu buffer'dan cek (AL'ye ascii kodu girer)
    int 16h          
    cmp al, ' '         ; Okunan tus SPACE mi?
    je sp_yes           ; Evetse sp_yes'e atla
    cmp ah, 39h         ; BIOS donanim kodu SPACE mi (garanti olsun diye)?
    je sp_yes
sp_no:
    xor al, al          ; AL'yi 0 yap (basilmadi anlami tasir)
    ret
sp_yes:
    mov al, 1           ; AL'yi 1 yap (basildi anlami tasir)
    ret
hizli_space_kontrol endp

; --- TEK KARAKTERI RENKLI YAZDIRMA (VRAM KULLANARAK) ---
renkli_kar proc
    ; Ekrana dogrudan VRAM'e yazar (cok hizlidir)
    ; AL = Yazilacak karakter, BL = Renk kodu, DH/DL = Satir/Sutun
    push ax 
    push bx 
    push cx 
    push dx 
    push di 
    push es
    push ax             ; Karakteri gecici olarak stack'e al
    
    ; VRAM offset (adres) hesabi: (Satir * 80 + Sutun) * 2
    mov al, 80          ; Satiri 80 ile carp (1 satir 80 karakter genisliginde)
    mul dh          
    mov dh, 0       
    add ax, dx          ; Sutunu ekle
    shl ax, 1           ; 2 ile carp (her hucre 2 byte: Karakter ve Renk)
    mov di, ax          ; Hesaplanan adresi DI registerina al
    
    mov cx, 0B800h      ; 0B800h adresi Color Monitor'lerin donanim bellek baslangicidir
    mov es, cx          ; ES registerine yukle
    
    pop ax              ; Karakteri (AL) geri al
    mov ah, bl          ; Renk kodunu (BL) AH registerina yukle (AX = Renk + Karakter oldu)
    mov es:[di], ax     ; VRAM bellekte hesaplanan adrese direkt bastir
    
    pop es 
    pop di 
    pop dx 
    pop cx 
    pop bx 
    pop ax
    ret
renkli_kar endp

; --- EKRANIN DEGISMIYEN KISIMLARINI CIZME ---
sabit_ekran_ciz proc
    ; --- KIRMIZI BASLIK YAZDIRMA ---
    mov dh, 1           ; 1. Satir
    mov dl, 28          ; 28. Sutun
    lea bp, baslik      ; Basligin adresi

    mov dl, 23
    call gotoxy
    lea dx, bilgi
    call print_dos

    mov dh, 3
    mov dl, 40
    mov al, 'V'
    mov bl, 0Eh         ; Sari renk
    call renkli_kar

    mov dh, 11
    mov dl, 40
    mov al, '+'
    mov bl, 0Fh         ; Beyaz renk
    call renkli_kar
    
    call cember_ciz     ; 'o' harflerinden olusan daireyi ciz

sabit_ekran_ciz endp

; --- NOKTALARDAN DAIRE CIZME ---
cember_ciz proc
    push ax 
    push bx 
    push cx 
    push dx 
    push si
    xor si, si          ; SI indeksini sifirla (Dizi okumak icin)
    mov cl, c_cnt       ; Dongu kac kere donecek (18 nokta var)
    xor ch, ch          ; CH'yi sifirla, tam CX hazirla
cb:
    mov dh, c_row[si]   ; Y koordinatini al
    mov dl, c_col[si]   ; X koordinatini al
    mov al, 'o'         ; Ekrana Basilacak harf 'o'
    mov bl, 08h         ; Koyu gri renk
    call renkli_kar     ; Noktayi ciz
    inc si              ; Bir sonraki koordinata gec
    loop cb             ; Donguyu devam ettir
    
    pop si 
    pop dx 
    pop cx 
    pop bx 
    pop ax
    ret
cember_ciz endp

; --- DONEN CARKI VE YAZILARI CIZME (OYUNUN ANA ANIMASYONU) ---
carki_ciz proc
    push ax 
    push bx 
    push cx 
    push dx 
    push si 
    push di 
    push es
    cld                 ; String okuma isleminde yonu soldan saga ayarla
    mov ax, 0B800h
    mov es, ax          ; VRAM bellek adresini ES'ye tanit
    
    ; 1) Ortadaki cizgi animasyonu
    xor bx, bx
    mov bl, spin_pos    ; Donme sekli indeksi (0, 1, 2, 3)
    mov al, spin_tab[bx]; Indekse gore sembolu al ('|', '/', '-', '\')
    mov ah, 0Fh         ; Parlak Beyaz renk kodu
    mov di, 1840        ; Merkezin (R11, C40) VRAM bellek adresi
    mov es:[di], ax     ; Sembolu direkt ekranin ortasina yapistir
    
    ; 2) Dilim Yazilarini Yerlestirme
    mov slot_tmp, 0     ; 8 ayri dilimi cizmek icin dongu sayaci (0'dan baslar)
draw_all_slots:
    mov al, cark_idx    ; Donen carkin o anki indexini AL'ye al
    add al, slot_tmp    ; O an cizilecek dilimin kacinci sirada oldugunu bul
    cmp al, 8           ; Eger 8'e ulastiysa (8 dilim var)
    jb idx_ready
    sub al, 8           ; 8 cikararak basa (0, 1..) sar. (Matematikteki Mod alma)
idx_ready:
    xor ah, ah          ; AH'yi sifirla ki AX tam bir rakam olsun
    mov bx, ax          ; Indeksi BX'e gecir (Dizi okumak icin zorunlu)
    shl bx, 1           ; Adresler word (2 byte) yer kapladigindan dizini 2 ile carp
    mov si, str_ptr[bx] ; O yazinin ('[ 100 ]' gibi) hafizadaki baslangic adresini al
    
    mov bx, ax          ; Renk tablosu byte byte (1 byte) oldugu icin carpmasiz al
    mov ah, col_val[bx] ; Bu yaziya ozel rengi AH (Renk Kodu) icine at
    
    xor bx, bx
    mov bl, slot_tmp    ; Su an ekranin neresine cizdigimizi tutan sayac
    shl bx, 1           ; VRAM adres dizisi de word boyutunda, yine 2 ile carp
    mov di, d_vram[bx]  ; Yazinin ekranda duracagi dogru VRAM adresini al
    
    mov cx, 7           ; Cizilecek stringin uzunlugu 7 harftir (ornegin '[ 100 ]')
draw_str:
    lodsb               ; SI'nin gosterdigi adresten 1 harf (byte) alip AL registerina koyar
    stosw               ; AX registerini (AH'da renk, AL'de harf var) VRAM'e yazar (di artar)
    loop draw_str       ; 7 karakter yazilana kadar kelime kelime devam et
    
    inc slot_tmp        ; Bir sonraki cark dilimini cizmek icin sayaci artir
    cmp slot_tmp, 8     ; 8 dilim bitti mi?
    jb draw_all_slots   ; Bitmediyse draw_all_slots etiketine don
    
    pop es 
    pop di 
    pop si 
    pop dx 
    pop cx 
    pop bx 
    pop ax
    ret
carki_ciz endp
